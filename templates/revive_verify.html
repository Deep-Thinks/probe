import hmac, hashlib, bisect
from math import comb

N = {{n_rows}}
MULTIPLIERS = [{{multipliers_str}}]
PMF = [comb(N, k) / 2**N for k in range(N + 1)]
CDF = [sum(PMF[:i+1]) for i in range(len(PMF))]

server_seed = "{{server_seed}}"
client_seed = "{{client_seed}}"
nonce       = {{nonce}}

digest = hmac.new(
    bytes.fromhex(server_seed),
    f"{client_seed}:{nonce}".encode("utf-8"),
    hashlib.sha256,
).digest()
rand_u = int.from_bytes(digest[:8], "big") / (1 << 64)
slot   = bisect.bisect_left(CDF, rand_u)
print("slot:", slot, "multiplier:", MULTIPLIERS[slot])
